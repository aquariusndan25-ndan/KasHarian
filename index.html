Pengaturan Sync & Google Sheets

Masukkan Web App URL yang Anda dapatkan dari peluncuran Google Apps Script di Google Sheet pribadi Anda.

Skrip Google Apps Script (`Kode.gs`)

Buka Google Sheets > Ekstensi > Apps Script, lalu tempelkan skrip di bawah ini: 

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var contents = JSON.parse(e.postData.contents);
    var transactions = contents.transactions || [];
    
    if (sheet.getLastRow() > 1) {
      sheet.getRange(2, 1, sheet.getLastRow() - 1, 6).clearContent();
    }
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["ID Transaksi", "Tanggal", "Tipe Transaksi", "Kategori", "Catatan", "Nominal (Rp)"]);
    }
    
    var rows = transactions.map(function(item) {
      return [
        item.id || "",
        item.date || "",
        item.type === 'penerimaan' ? 'Penerimaan (Masuk)' : 'Pengeluaran (Keluar)',
        item.category || "",
        item.description || "",
        Number(item.amount) || 0
      ];
    });
    
    if (rows.length > 0) {
      sheet.getRange(2, 1, rows.length, 6).setValues(rows);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: rows.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var lastRow = sheet.getLastRow();
    var transactions = [];
    
    if (lastRow > 1) {
      var data = sheet.getRange(2, 1, lastRow - 1, 6).getValues();
      transactions = data.map(function(row) {
        return {
          id: String(row[0]),
          date: row[1] instanceof Date ? row[1].toISOString().split('T')[0] : String(row[1]),
          type: String(row[2]).toLowerCase().indexOf('penerimaan') !== -1 ? 'penerimaan' : 'pengeluaran',
          category: String(row[3]),
          description: String(row[4]),
          amount: Number(row[5])
        };
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", transactions: transactions }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", transactions: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
